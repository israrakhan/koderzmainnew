/* Theme Color Palette Reference */

/* Cream & White Theme Colors */

/* Primary Colors */
Primary Cream: #f5f3f0 - Main background, light surfaces
Primary White: #ffffff - Pure white for cards, overlays
Dark Cream: #ede9e0 - Slightly darker variant for hover
Light Cream: #faf8f5 - Slightly lighter variant

/* Accent Colors (Warm Browns & Tans) */
Warm Accent: #d4a574 - Primary accent, buttons, highlights
Warm Dark: #b8956f - Darker accent for active states
Warm Light: #e6c8a0 - Lighter accent for borders
Brown: #8b7355 - Text on light backgrounds

/* Text Colors */
Primary Text: #3c3c3c - Main body text
Secondary Text: #666666 - Subtexts, descriptions
Light Text: #999999 - Disabled text, hints
White Text: #ffffff - Text on colored backgrounds

/* Borders & Shadows */
Border: #e8dfd5 - Subtle borders
Border Light: #f0e8df - Very light borders
Shadow Light: rgba(139, 115, 85, 0.08) - Subtle shadows
Shadow Medium: rgba(139, 115, 85, 0.12) - Medium shadows

/* Status Colors */
Success: #4caf50 - Green for success states
Error: #dc3545 - Red for errors, danger
Warning: #ff9800 - Orange for warnings
Info: #2196f3 - Blue for information

/* Gradients */
Primary Gradient: linear-gradient(135deg, #f5f3f0 0%, #faf8f5 100%)
Accent Gradient: linear-gradient(135deg, #d4a574 0%, #b8956f 100%)
Warm Gradient: linear-gradient(135deg, #e6c8a0 0%, #d4a574 100%)

/* Usage in Components */
.navbar - Cream + White gradient background with warm accents
.nav-link - Warm accent with brown text
.nav-link:hover - Increased warm accent opacity
.nav-link.active - Warm accent gradient with white text
.logout-link - Error red with transparency
